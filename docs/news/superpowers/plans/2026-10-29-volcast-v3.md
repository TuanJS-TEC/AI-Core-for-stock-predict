```python
def test_dividend_seam_fixes_live_ret1_only():
    df = _bars(30)
    df["ref"] = np.nan
    a = df.index[(df.symbol == "AAA")][-1]           # last AAA row = a live ex-dividend day
    df.loc[a, "close"] = df.loc[a - 1, "close"] * 0.7 # raw close vs unadjusted prior: -30%
    df.loc[a, "ref"] = df.loc[a, "close"] / 1.01      # exchange ref (adjusted prior close): +1% on the day
    b = df.index[(df.symbol == "BBB")][-1]            # history row (ref NaN) with a real -30% move

    assert out.loc[("AAA", df.loc[a, "date"]), "ret1"] == pytest.approx(0.01)
    assert out.loc[("BBB", df.loc[b, "date"]), "ret1"] == pytest.approx(-0.3)


def test_dividend_seam_ignores_missing_or_zero_ref():
```

```python
def test_cross_sectional_ranks_are_per_day_and_bounded():
    df = _bars(40)
    lone = df[(df.symbol == "AAA")].index[-1]              # AAA alone on its last day
    df = df.drop(df[(df.symbol == "BBB") & (df.date == df.loc[lone, "date"])].index)
    out = volcast.add_features(df, NO_BCTC).dropna(subset=["cs_vol10", "cs_vol20"])   # past warm-up
    for c in ("cs_vol10", "cs_vol20", "cs_range", "cs_abs_ret1"):
        assert out[c].between(0, 1, inclusive="neither").all()
        two = out.groupby("date")[c].transform("count") == 2
        assert set(out.loc[two, c].round(6)) <= {0.25, 0.5, 0.75}   # 0.5 = tie (_bars gives every ticker range 0.02)
    assert out[out.date == df.loc[lone, "date"]].cs_vol10.tolist() == [0.5]


def test_vol_ratio_is_one_for_a_flat_ticker():
    df = _bars(40)
    df.loc[df.symbol == "AAA", ["close", "high", "low"]] = 10.0
    out = volcast.add_features(df, NO_BCTC)
    assert (out[out.symbol == "AAA"].vol_ratio.dropna() == 1.0).all()
    assert out[out.symbol == "AAA"].vol_ratio.notna().any()
```

  còn lại của `walk` vẫn đọc `best = (m, va_acc, t, va_cov)` như cũ.
